def para(text, size=10.5, leading=15, color=INK, space=10, bold=False):
    global Y
    style = ParagraphStyle(
        'body', fontName='Helvetica-Bold' if bold else 'Helvetica',
        fontSize=size, leading=leading, textColor=color,
    )
    p = Paragraph(text, style)
    _, height = p.wrap(CONTENT, H)
    if Y - height < 58:
        raise ValueError(f'Teks melewati area halaman pada y={Y}: {text[:70]}')
    p.drawOn(C, M, Y - height)
    Y -= height + space


def heading(text):
    para(escape(text), size=13, leading=18, color=TEAL, bold=True, space=10)


def begin(number, title, subtitle=''):

        Y -= 12

    C.setFillColor(MUTED)
    C.setFont('Helvetica', 8)
    C.drawString(M, 30, 'Muhamad Husni Al Muhjiri | 10 Oktober 2026')
    C.drawRightString(W - M, 30, f'{number} / 7')


def table(headers, rows, widths, size=9):
    global Y
    style = ParagraphStyle('table', fontName='Helvetica', fontSize=size, leading=size + 4, textColor=INK)
    for index, row in enumerate([headers] + rows):
        cells = []
        height = 0
        for text, width in zip(row, widths):
            cell = Paragraph(escape(str(text)), style)
            _, cell_height = cell.wrap(width - 16, H)
            cells.append((cell, cell_height))
            height = max(height, cell_height + 16)
        if Y - height < 58:
            raise ValueError('Tabel melewati area halaman')
        C.setFillColor(PALE if index == 0 else (colors.HexColor('#F8FAFC') if index % 2 == 0 else colors.white))
        C.rect(M, Y - height, CONTENT, height, fill=1, stroke=0)
        x = M
        for (cell, cell_height), width in zip(cells, widths):
            cell.drawOn(C, x + 8, Y - 8 - cell_height)
            x += width
        C.setStrokeColor(LINE)
        C.line(M, Y - height, W - M, Y - height)
        Y -= height
    Y -= 18


def pair(left_file, left_label, right_file, right_label):
    global Y
    gap = 20
    width = (CONTENT - gap) / 2
    height = width * 844 / 390
    for x, filename, label in [(M, left_file, left_label), (M + width + gap, right_file, right_label)]:
        C.setFillColor(INK)
        C.setFont('Helvetica-Bold', 11)
        C.drawString(x, Y - 4, label)
        C.setStrokeColor(LINE)
        C.rect(x - 0.5, Y - 21 - height - 0.5, width + 1, height + 1, fill=0, stroke=1)
        C.drawImage(str(SHOTS / filename), x, Y - 21 - height, width=width, height=height)
    Y -= height + 37


def code(lines):
    global Y
    height = len(lines) * 15 + 24
    C.setFillColor(PALE)
    C.roundRect(M, Y - height, CONTENT, height, 10, fill=1, stroke=0)
    C.setFillColor(INK)
    C.setFont('Courier', 10)
    for line in lines:
        C.drawString(M + 12, Y - 20, line)
        Y -= 15
    Y -= 34

# Form kelas harus tetap interaktif dan mempunyai field kanonis serta widget.
reader = PdfReader(str(OUTPUT))
assert len(reader.pages) == 7
fields = reader.get_fields()
assert fields and 'kelas_mahasiswa' in fields
expected_class = IDENTITY.get('kelas') or ''
assert str(fields['kelas_mahasiswa'].get('/V', '')) == expected_class
widgets = []
for page in reader.pages:
    for ref in page.get('/Annots', []):
        widget = ref.get_object()
        if widget.get('/Subtype') == '/Widget':
            widgets.append(widget)
assert len(widgets) == 1
assert str(widgets[0].get('/V', '')) == expected_class
assert widgets[0].get('/AP') and widgets[0]['/AP'].get('/N')
assert all(page.extract_text().strip() for page in reader.pages)
print(f'PDF siap: {OUTPUT} (7 halaman, kolom kelas interaktif terverifikasi).')
